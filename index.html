Kode Google Apps Script (doPost & doGet)
  1. Buka Google Sheets baru > menu Extensions > Apps Script.
  2. Tempel kode di bawah, lalu klik Deploy > New deployment > Web app (Akses: Anyone).
  3. Salin Web app URL ke kolom di atas (atau tanam langsung di variabel DEFAULT_GSHEET_URL dalam file HTML sebelum di-deploy ke Vercel).
function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["ID Order", "Waktu Order", "Nama Pemesan", "Divisi / Lantai", "Metode Penyerahan", "Rincian Pesanan", "Catatan / Varian Cadangan", "Total Item", "Estimasi Berat (kg)", "Total Harga (Rp)", "Items JSON"]);
  }
  var data = JSON.parse(e.postData.contents);
  sheet.appendRow([
    data.id,
    data.timestamp,
    data.name,
    data.division,
    data.pickupMethod,
    data.itemsSummary,
    data.notes || "-",
    data.totalQty,
    data.totalWeightKg,
    data.totalPrice,
    JSON.stringify(data.items || [])
  ]);
  return ContentService.createTextOutput(JSON.stringify({status: "success"})).setMimeType(ContentService.MimeType.JSON);
}

function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var rows = sheet.getDataRange().getValues();
  var orders = [];
  for (var i = 1; i < rows.length; i++) {
    var r = rows[i];
    var parsedItems = [];
    try { parsedItems = JSON.parse(r[10] || "[]"); } catch(err) {}
    orders.push({
      id: r[0],
      timestamp: r[1],
      name: r[2],
      division: r[3],
      pickupMethod: r[4],
      itemsSummary: r[5],
      notes: r[6] === "-" ? "" : r[6],
      totalQty: Number(r[7]) || 0,
      totalWeightKg: Number(r[8]) || 0,
      totalPrice: Number(r[9]) || 0,
      items: parsedItems
    });
  }
  return ContentService.createTextOutput(JSON.stringify({status: "success", orders: orders.reverse()})).setMimeType(ContentService.MimeType.JSON);
}
